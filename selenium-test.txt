Prueba HTML CON SELENIUM.

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;

public class TestProyectoGil {

    public static void main(String[] args) {

        System.setProperty("webdriver.chrome.driver", "C:\\drivers\\chromedriver.exe");

        WebDriver driver = new ChromeDriver();

        // URL de tu proyecto
        driver.get("https://proyectogillll.42web.io/ProyectoGil/index.html");

        // Llenar formulario
        driver.findElement(By.name("nombre")).sendKeys("Juan");
        driver.findElement(By.name("puesto")).sendKeys("Developer");
        driver.findElement(By.name("horas")).sendKeys("8");

        // Enviar formulario
        driver.findElement(By.tagName("button")).click();

        // Validar respuesta
        String resultado = driver.getPageSource();

        if (resultado.contains("correcto") || resultado.contains("Registro")) {
            System.out.println("Prueba exitosa");
        } else {
            System.out.println("Error en la prueba");
        }

        driver.quit();
    }
}
